System.Action<UnityEngine.GameObject> trackOwned = delegate(UnityEngine.GameObject root) {
    foreach (var transform in root.GetComponentsInChildren<UnityEngine.Transform>(true))
        if (seenOwnedIds.Add(transform.gameObject.GetInstanceID())) ownedObjects.Add(transform.gameObject);
    var meshes = new System.Collections.Generic.List<UnityEngine.Mesh>();
    foreach (var filter in root.GetComponentsInChildren<UnityEngine.MeshFilter>(true)) if (filter.sharedMesh != null) meshes.Add(filter.sharedMesh);
    foreach (var renderer in root.GetComponentsInChildren<UnityEngine.SkinnedMeshRenderer>(true)) if (renderer.sharedMesh != null) meshes.Add(renderer.sharedMesh);
    foreach (var mesh in meshes)
    {
        if (UnityEditor.EditorUtility.IsPersistent(mesh)) { rememberPersistent(mesh); continue; }
        if (!borrowedTransientIds.Contains(mesh.GetInstanceID()) && seenOwnedIds.Add(mesh.GetInstanceID())) ownedMeshes.Add(mesh);
    }
    foreach (var renderer in root.GetComponentsInChildren<UnityEngine.Renderer>(true))
        foreach (var material in renderer.sharedMaterials)
        {
            if (material == null) continue;
            if (UnityEditor.EditorUtility.IsPersistent(material)) { rememberPersistent(material); continue; }

// All generated meshes are private to the factory root. Clear its production

System.Action<UnityEngine.GameObject> destroyOwned = delegate(UnityEngine.GameObject go) {
    if (go == null) return;
    trackOwned(go);
    foreach (var component in go.GetComponentsInChildren<UnityEngine.MonoBehaviour>(true))
    {
        if (component == null || component.GetType().Name != "GeneratedMeshOwner") continue;
        var field = component.GetType().GetField("_meshes", instanceFlags);
        var meshes = field == null ? null : field.GetValue(component) as System.Collections.IList;
        if (meshes == null) continue;
        var owned = new System.Collections.Generic.List<UnityEngine.Mesh>();
        foreach (var item in meshes) { var mesh = item as UnityEngine.Mesh; if (mesh != null) owned.Add(mesh); }
        meshes.Clear();
        foreach (var mesh in owned)
        {
            if (UnityEditor.AssetDatabase.Contains(mesh)) throw new System.InvalidOperationException("Factory ownership incorrectly includes a persistent mesh");
            UnityEngine.Object.DestroyImmediate(mesh);
        }
    }
    UnityEngine.Object.DestroyImmediate(go);
};

    }
    var cornerRows = new Newtonsoft.Json.Linq.JArray();
    var radius = 0.0;
    foreach (var point in corners(bounds)) { cornerRows.Add(vector(point)); radius = System.Math.Max(radius, System.Math.Sqrt(point.x * point.x + point.z * point.z)); }

        ["prototypeId"] = prototype, ["sourceArtId"] = definition.ArtId, ["wearBundleId"] = bundleId,

                    var low = new UnityEngine.Vector3((float)minimum["x"], (float)minimum["y"], (float)minimum["z"]);
                    var high = new UnityEngine.Vector3((float)maximum["x"], (float)maximum["y"], (float)maximum["z"]);
                    if (!hasMerged) { merged = new UnityEngine.Bounds(low, UnityEngine.Vector3.zero); hasMerged = true; }
